import      { afterEach, expect, test } from "bun:test"  ;
import      {
              chmodSync             ,
              mkdirSync             ,
              mkdtempSync           ,
              readFileSync          ,
              realpathSync          ,
              rmSync                ,
              symlinkSync           ,
              writeFileSync         ,
                                      } from "node:fs"   ;
import      { tmpdir                  } from "node:os"   ;
import      { dirname, join, resolve  } from "node:path" ;

const harness         = await import(join(import.meta.dir, "../development/tools/release-hud-lifecycle.mjs")) ;
const roots: string[] = []                                                                                    ;
function temporary() { const root = mkdtempSync(join(tmpdir(), "boxbox-hud-lifecycle-unit-")); roots.push(root); return root; }
afterEach(() => { for (const root of roots.splice(0)) rmSync(root, { recursive: true, force: true }); });

function coreFixture(host: "codex" | "claude" = "codex") {
	const root = temporary(); const output = join(root, "core"); const workspace = join(output, "workspace-real");
	const previous = join(workspace, "data", "boxbox", host, "0.2.3"); const current = join(workspace, "data", "boxbox", host, "0.2.4");
	mkdirSync(previous, { recursive: true }); mkdirSync(current, { recursive: true });
	const path = join(output, "lifecycle.json");
	const report = {
		schemaVersion: 1, ok: true, host, platform: process.platform === "linux" ? "linux" : "darwin", workspace,
		artifacts : { previous: { version: harness.HUD_ARTIFACTS.previous.version, sha256: harness.HUD_ARTIFACTS.previous.archiveSha256 }, current: { version: harness.HUD_ARTIFACTS.current.version, sha256: harness.HUD_ARTIFACTS.current.archiveSha256 } },
		steps     : [{ label: "final-remove", registrationRemoved: true }],
		residue   : [{ role: "previous", path: previous, present: true }, { role: "current", path: current, present: true }],
	};
	writeFileSync(path, JSON.stringify(report)); return { root, output, workspace, previous, current, path, report };
}

test("HUD lifecycle arguments accept only one core report and a new output", () => {
	expect(harness.parseArguments(["--core", "core.json", "--output", "evidence"])).toEqual({ core: resolve("core.json"), output: resolve("evidence") });
	for (const args of [[], ["--core", "a"], ["--core", "a", "--output", "b", "--hud"], ["--core", "a", "--core", "b", "--output", "c"], ["--core", "same", "--output", "same"]]) expect(() => harness.parseArguments(args)).toThrow();
});

test("core prerequisite binds exact artifacts, final removal, canonical contained residue and supported OS", () => {
	const f = coreFixture();
	expect(harness.validateCoreReport(f.report, f.path)).toMatchObject({ host: "codex", workspace: realpathSync(f.workspace), roots: { previous: realpathSync(f.previous), current: realpathSync(f.current) } });
	for (const mutate of [
		(r: any) => { r.ok = false; },
		(r: any) => { r.platform = "win32"; },
		(r: any) => { r.steps = []; },
		(r: any) => { r.artifacts.current.sha256 = "wrong"; },
		(r: any) => { r.residue[1].present = false; },
	]) {
		const changed = structuredClone(f.report); mutate(changed); expect(() => harness.validateCoreReport(changed, f.path)).toThrow();
	}
	const outside = temporary(); const changed = structuredClone(f.report); changed.residue[1].path = outside; expect(() => harness.validateCoreReport(changed, f.path)).toThrow("outside");
});

test("Codex launcher expectation quotes hostile paths and exact ownership rejects edits and symlinks", () => {
	const root = temporary(); const plugin = join(root, "plugin ' $(touch never)"); const node = join(root, "node ' binary");
	const expected = harness.expectedCodexLauncher({ node }, plugin); const path = join(root, "boxbox-codex"); writeFileSync(path, expected); chmodSync(path, 0o755);
	expect(expected).toContain("'\\''"); expect(expected).toContain(' -- "$@"'); expect(() => harness.assertExactOwned(path, expected)).not.toThrow();
	chmodSync(path, 0o644); expect(() => harness.assertExactOwned(path, expected)).toThrow("mode"); chmodSync(path, 0o755);
	writeFileSync(path, expected + "# edit\n"); expect(() => harness.assertExactOwned(path, expected)).toThrow("changed");
	const target = join(root, "target"); writeFileSync(target, expected); chmodSync(target, 0o755); const link = join(root, "link"); symlinkSync(target, link); expect(() => harness.assertExactOwned(link, expected)).toThrow("nonsymlink");
});

test("Claude semantic preservation removes only statusLine", () => {
	const settings = { env: { KEEP: "yes" }, permissions: { defaultMode: "default" }, statusLine: { type: "command", command: "owned" } };
	expect(harness.withoutStatusLine(settings)).toEqual({ env: { KEEP: "yes" }, permissions: { defaultMode: "default" } });
	expect(settings.statusLine.command).toBe("owned");
});

test("isolated environment validates only declared path keys and ignores scalar flags", () => {
	const workspace                           = temporary()                                              ;
	const environment: Record<string, string> = { PATH: process.env.PATH ?? "", NO_COLOR: "1", CI: "1" } ;
	for (const key of harness.ISOLATED_PATH_ENVIRONMENT_KEYS) {
		environment[key] = join(workspace, key.toLowerCase());
		mkdirSync(environment[key], { recursive: true });
	}

	expect(harness.validateEnvironmentPaths(environment, workspace))    .toEqual  ([...harness.ISOLATED_PATH_ENVIRONMENT_KEYS]) ;
	expect(harness.ISOLATED_PATH_ENVIRONMENT_KEYS                  ).not.toContain("NO_COLOR"                                 ) ;
	expect(harness.ISOLATED_PATH_ENVIRONMENT_KEYS                  ).not.toContain("CI"                                       ) ;
	expect(harness.ISOLATED_PATH_ENVIRONMENT_KEYS                  ).not.toContain("PATH"                                     ) ;

	const outside = temporary();
	expect(() => harness.validateEnvironmentPaths({ ...environment, CODEX_HOME: outside }, workspace)).toThrow("outside");
	expect(() => harness.validateEnvironmentPaths({ ...environment, HOME: undefined }, workspace)).toThrow("Missing isolated path environment variable: HOME");
});


test("non-following inventory detects unrelated mutations and records symlink targets", () => {
	const root = temporary(); const unrelated = join(root, "unrelated.txt"); const target = join(root, "target"); const link = join(root, "link");
	writeFileSync(unrelated, "before"); writeFileSync(target, "target"); symlinkSync("target", link);
	const before = harness.inventoryScopes([root]);
	writeFileSync(unrelated, "after"); const after = harness.inventoryScopes([root]);
	expect(() => harness.assertInventoryDelta(before, after, [harness.inventoryKey(root, "owned.txt")])     ).toThrow     ("Unexpected inventory mutation") ;
	expect(harness.assertInventoryDelta(before, after, [harness.inventoryKey(root, "unrelated.txt")])       ).toHaveLength(1                              ) ;
	expect(Object.values(before).some((entry: any) => entry.kind === "symlink" && entry.target === "target")).toBe        (true                           ) ;

	const other = temporary();
	writeFileSync(join(other, "unrelated.txt"), "before");
	const twoBefore = harness.inventoryScopes([root, other]);
	writeFileSync(join(other, "unrelated.txt"), "after");
	const twoAfter = harness.inventoryScopes([root, other]);
	expect(() => harness.assertInventoryDelta(twoBefore, twoAfter, [harness.inventoryKey(root, "unrelated.txt")])).toThrow("Unexpected inventory mutation");
	expect(harness.assertInventoryDelta(twoBefore, twoAfter, [harness.inventoryKey(other, "unrelated.txt")])).toHaveLength(1);
});

test("residue roots and setup files reject symlink identity", () => {
	const f = coreFixture(); const residueLink = join(f.workspace, "linked-residue"); symlinkSync(f.previous, residueLink, process.platform === "win32" ? "junction" : "dir");
	const linkedReport = structuredClone(f.report); linkedReport.residue[0].path = residueLink;
	expect(() => harness.validateCoreReport(linkedReport, f.path)).toThrow("nonsymlink directory");

	const plugin = join(f.previous, "plugin"); const scripts = join(plugin, "scripts", "hud"); mkdirSync(scripts, { recursive: true });
	const target = join(f.previous, "setup-target.mjs"); writeFileSync(target, "export {};"); symlinkSync(target, join(scripts, "setup.mjs"));
	writeFileSync(join(plugin, "runtime.json"), "{}");
	expect(() => harness.validateBinding("previous", f.previous, f.workspace)).toThrow("regular nonsymlink file");
});

test("ownership refusal requires the pinned diagnostic class", () => {
	expect(harness.assertOwnershipRefusal("[BoxBox] Another file or a user edit owns /tmp/bin", "codex", false)                       ).toContain("user edit"                   ) ;
	expect(harness.assertOwnershipRefusal("[BoxBox] Claude already has a statusLine. Keep it", "claude", false)                       ).toContain("already has"                 ) ;
	expect(harness.assertOwnershipRefusal("[BoxBox] Claude statusLine is not owned by this HUD; no settings changed.", "claude", true)).toContain("not owned"                   ) ;
	expect(() => harness.assertOwnershipRefusal("SyntaxError: unrelated exit 1", "codex", false)                                      ).toThrow  ("ownership refusal diagnostic") ;
});

test("failed core prerequisite retains a report and executes no HUD command", async () => {
	const root = temporary(); const core = join(root, "core.json"); const output = join(root, "evidence"); writeFileSync(core, JSON.stringify({ schemaVersion: 1, ok: false }));
	await expect(harness.hudLifecycle({ core, output })).rejects.toThrow("did not pass");
	const report = JSON.parse(readFileSync(join(output, "hud-lifecycle.json"), "utf8"));
	expect(report.ok).toBe(false); expect(report.commands).toEqual([]); expect(report.error).toContain("did not pass");
	await expect(harness.hudLifecycle({ core, output })).rejects.toThrow("existing runs");
});
